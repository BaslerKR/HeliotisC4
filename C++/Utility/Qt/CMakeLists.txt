cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(HeliotisC4Presentation LANGUAGES CXX)
endif()
if(TARGET HeliotisC4Presentation)
    return()
endif()
if(TARGET Qt6::Widgets)
    set(presentation_qt Qt6)
elseif(TARGET Qt5::Widgets)
    set(presentation_qt Qt5)
else()
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    set(presentation_qt Qt6)
endif()
if(NOT TARGET Resources::Widgets)
    message(FATAL_ERROR "[HeliotisC4] The presentation requires the Qt-only Resources::Widgets target.")
endif()
add_library(HeliotisC4Presentation STATIC
    HeliotisC4ControlUi.cpp HeliotisC4ControlUi.h HeliotisC4FeatureTree.cpp HeliotisC4FeatureTree.h)
set_target_properties(HeliotisC4Presentation PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
target_compile_features(HeliotisC4Presentation PUBLIC cxx_std_17)
target_compile_options(HeliotisC4Presentation PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_include_directories(HeliotisC4Presentation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(HeliotisC4Presentation PUBLIC ${presentation_qt}::Widgets Resources::Widgets)
